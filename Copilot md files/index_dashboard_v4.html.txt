        function switchEdaTab(tabName) {
            ['outliers', 'heatmap', 'counts', 'pairplot'].forEach(t => {

                document.getElementById(`tab-${t}`).className = "px-3 py-1.5 rounded-lg font-medium bg-slate-700 text-slate-300 hover:bg-slate-600 transition";
            });
            document.getElementById(`eda-${tabName}`).classList.remove('hidden');
            document.getElementById(`tab-${tabName}`).className = "px-3 py-1.5 rounded-lg font-medium bg-indigo-600 text-white transition";
        }
    </script>
